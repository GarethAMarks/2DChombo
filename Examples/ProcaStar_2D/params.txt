# Perturbations. With proca_perturb = 1 the azimuthal index proca_perturb_m selects the perturbation:
#   m = 0: the spherical perturbation (f, g) -> (1 + a exp(-(r - r0)^2 / w^2)) (f, g), with the metric and E (Gauss law) re-solved, as in
#          Spherical_BS_Evolver (a = proca_perturb_amp, w = proca_perturb_spread, r0 = proca_perturb_center);
#   m >= 1 (needs proca_perturb_aspherical = 1): the aspherical perturbation of arXiv:2609.30021, Eq. (11): all matter
#          fields multiplied by exp(-2 delta cos(m phi)), delta = proca_perturb_amp, phi the polar angle from the
#          symmetry axis about the star centre. In 2D it is axisymmetric; the constraints are not re-solved. Odd m is
#          incompatible with a reflective x boundary. Single stars only.
# proca_perturb_aspherical = 0 disables the aspherical perturbations (the perturbation is then always spherical).
# For m = 0, a > 0 makes the star denser. For m >= 1 the Gauss constraint still holds for a star at rest (E^i is radial
# and the factor depends only on angles), except next to the centre, where phi is undefined.
proca_perturb = 0
proca_perturb_aspherical = 0
proca_perturb_m = 0
proca_perturb_amp = 0.01
proca_perturb_spread = 3.
proca_perturb_center = 0.
